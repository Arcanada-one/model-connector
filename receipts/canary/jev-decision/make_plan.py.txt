#!/usr/bin/env python3
"""Build CanaryPlan/v2 + MeasuredGitSource/v1 for the model-connector JEV decision adapter. argv: SUBJECT OUT"""
import hashlib, json, subprocess, sys
from datetime import datetime, timezone
from pathlib import Path
REPO, OUT = Path(sys.argv[1]).resolve(), Path(sys.argv[2]).resolve()
OUT.mkdir(parents=True, exist_ok=True)
HERE = Path(__file__).resolve().parent
PY = '/usr/bin/python3.12'
sha = lambda p: hashlib.sha256(Path(p).read_bytes()).hexdigest()
git = lambda *a: subprocess.check_output(['git', '-C', str(REPO), *a], text=True).strip()
assert not git('status', '--porcelain'), 'subject must be clean'
commit, tree = git('rev-parse', 'HEAD'), git('rev-parse', 'HEAD^{tree}')
evidence = {'schema': 'MeasuredGitSource/v1', 'commit': commit, 'tree': tree, 'dirty': False,
            'producer': {'tool': 'control-jev-mc/canary/make_plan.py', 'version': '1.0.0'},
            'captured_at_utc': datetime.now(timezone.utc).strftime('%Y-%m-%dT%H:%M:%SZ')}
ev_raw = json.dumps(evidence, indent=1, sort_keys=True).encode() + b'\n'
(OUT/'source-evidence.json').write_bytes(ev_raw)
files = [HERE/'harness.py', HERE/'stub.mjs', HERE/'driver.cjs', Path('/usr/bin/node'),
         REPO/'src/connectors/jev/jev.connector.ts', REPO/'src/connectors/jev/decision.contract.ts',
         REPO/'src/connectors/base-api.connector.ts']
files += [REPO/'dist/src/connectors'/p for p in ('jev/jev.connector.js', 'jev/decision.contract.js',
                                                 'jev/decision.fixture.js', 'base-api.connector.js')]
manifest = [{'path': str(p), 'sha256': sha(p)} for p in files]
probe = {'kind': 'process', 'id': 'loopback-tls-stub-native-decision', 'argv': [PY, str(HERE/'harness.py')], 'cwd': '.',
         'env_keys': ['PATH', 'TMPDIR'], 'timeout_seconds': 120, 'capture_limit_bytes': 262144,
         'executable_sha256': sha(PY), 'execution_files': manifest,
         'expect': {'exit_codes': [0], 'stdout_contains': [
             'CANARY ok private network namespace: external network unreachable',
             'CANARY ok POST https://api.typesafe.ai/v1/systemone over TLS to loopback stub; shadow decision observed, answers and model preserved',
             'CANARY ok oversize-chunked (>128 KiB) body refused as JEV parse_error',
             'CANARY ok oversize-length (>128 KiB) body refused as JEV parse_error',
             'CANARY ok 302 redirect not followed (1 request)',
             'CANARY ok HTTP 500 body not echoed',
             'CANARY ok out-of-domain answer not observed',
             'CANARY ok JEV_ENABLED=false stops the next call before any request (kill switch re-read)',
             'CANARY ok all scenarios: responses never contain the key or an upstream body',
             'CANARY ok cleanup throwaway dir removed']},
         'entities': ['code_unit:src/connectors/jev/jev.connector.ts'], 'mutating': True}
plan = {'schema': 'CanaryPlan/v2', 'id': 'model-connector-jev-decision-loopback', 'environment': 'arcana-devs-netns-loopback',
        'owner': 'Control network hand for JEV-audit remediation (Arcanada, claude-opus-5-5) on arcana-devs', 'read_only': False,
        'subject': {'schema': 'GitCanarySubject/v1', 'commit': commit, 'tree': tree,
                    'evidence': {'path': 'source-evidence.json', 'sha256': hashlib.sha256(ev_raw).hexdigest(), 'source_field': 'commit'}},
        'purpose': ('AUP-GRAPH-008 canary for code_unit:src/connectors/jev/jev.connector.ts on its inferred provider boundary, '
                    'measured on a local candidate without real provider keys: the built subject connector runs with real fetch '
                    'and TLS inside a private user/network/mount namespace where api.typesafe.ai resolves to a loopback HTTPS stub '
                    'signed by a throwaway CA and the key is a random dummy. Asserts endpoint/method/auth header, preserved shadow '
                    'decision, >128 KiB bodies refused (valid padded decisions, so only the byte bound can refuse), redirect not '
                    'followed, error bodies and key never echoed, out-of-domain answers unknown, kill switch re-read. Mutant with '
                    'the byte bound removed is red (see mutant-unbounded.log.txt). Not a measurement of the real provider.'),
        'probes': [probe]}
(OUT/'plan.json').write_text(json.dumps(plan, indent=1) + '\n')
print('commit', commit)
